    function buildColourPalette() {

        const paletteGroups =
            document.querySelectorAll('[data-family]');

        paletteGroups.forEach(function(group) {

            const family = group.dataset.family;

            let colours = [];

            if (family === 'neutral') {

                colours = [
                    { name: 'black', css: 'var(--color-black)' },
                    { name: 'white', css: 'var(--color-white)' }
                ];

            } else {

                colours = paletteColours.filter(function(colour) {
                    return colour.name.startsWith(family + '-');
                });

            }

            colours.forEach(function(colour) {

                const button = document.createElement('button');

                button.type = 'button';
                button.className = 'colour-swatch palette-swatch';

                button.dataset.colour = colour.name;

                button.style.backgroundColor = colour.css;

                    'aria-label',

                button.addEventListener('keydown', handlePaletteKeyboard);

        });

    }

                break;

            case 'start':
                settings.gradientStart = colourName;
                break;

            case 'end':
                settings.gradientEnd = colourName;
                break;

            case 'h1':
                settings.h1Colour = colourName;
                break;

            case 'h2':
                settings.h2Colour = colourName;
                break;

        }

        updatePreview();

    }


    // ==========================================
    // 8. GET CURRENT COLOUR
    // ==========================================

    function getActiveColour() {

        switch (settings.activeTarget) {

            case 'overlay':
                return settings.solidColour;

            case 'start':
                return settings.gradientStart;

            case 'end':
                return settings.gradientEnd;

            case 'h1':
                return settings.h1Colour;

            case 'h2':
                return settings.h2Colour;

            default:
                return 'white';

        }

    }


    // ==========================================
    // 9. UPDATE PALETTE SELECTION
    // ==========================================

    function updatePaletteSelection() {

        const currentColour = getActiveColour();

        const buttons = document.querySelectorAll(
            '.palette-swatch'
        );

        buttons.forEach(function(button) {

            button.setAttribute(
                'aria-pressed',
                String(button.dataset.colour === currentColour)
            );

        });

    }


    // ==========================================
    // 10. PALETTE KEYBOARD SUPPORT
    // ==========================================

    function handlePaletteKeyboard(event) {

        const allowedKeys = [
            'ArrowRight',
            'ArrowLeft',
            'ArrowUp',
            'ArrowDown',
            'Home',
            'End'
        ];

        if (!allowedKeys.includes(event.key)) {
            return;
        }

        const buttons = Array.from(
            document.querySelectorAll('.palette-swatch')
        );

        const currentIndex = buttons.indexOf(event.currentTarget);

        let nextIndex = currentIndex;

        if (event.key === 'ArrowRight' || event.key === 'ArrowDown') {
            nextIndex = Math.min(currentIndex + 1, buttons.length - 1);
        }

        if (event.key === 'ArrowLeft' || event.key === 'ArrowUp') {
            nextIndex = Math.max(currentIndex - 1, 0);
        }

        if (event.key === 'Home') {
            nextIndex = 0;
        }

        if (event.key === 'End') {
            nextIndex = buttons.length - 1;
        }

        event.preventDefault();

        buttons[nextIndex].focus();

    }


    // ==========================================
    // 11. READ SETTINGS
    // ==========================================

    function readSettings() {

        settings.centreText =
            centreTextCheckbox.checked;

        settings.minimumHeight =
            minimumHeightCheckbox.checked;

        settings.showSubheading =
            showSubheadingCheckbox.checked;

        settings.gradientOverlay =
            gradientOverlayCheckbox.checked;

        settings.overlayOpacity =
            Number(overlayOpacitySlider.value);

        settings.gradientDirection =
            gradientDirection.value;

        settings.h1Font = h1Font.value;
        settings.h2Font = h2Font.value;

        settings.h1Spacing = Number(h1Spacing.value);
        settings.h2Spacing = Number(h2Spacing.value);

    }


    // ==========================================
    // 12. UPDATE OVERLAY
    // ==========================================

    function updateOverlay() {

        heroOverlay.style.opacity =
            settings.overlayOpacity;

        if (settings.gradientOverlay) {

            heroOverlay.classList.remove('hero-overlay-solid');
            heroOverlay.classList.add('hero-overlay-gradient');

            const startColour = paletteColours.find(
                colour => colour.name === settings.gradientStart
            );

            const endColour = paletteColours.find(
                colour => colour.name === settings.gradientEnd
            );

            heroOverlay.style.backgroundImage =
                `linear-gradient(${settings.gradientDirection}, ${startColour.css}, ${endColour.css})`;

            heroOverlay.style.backgroundColor = 'transparent';

        } else {

            heroOverlay.classList.remove('hero-overlay-gradient');
            heroOverlay.classList.add('hero-overlay-solid');

            const solidColour = paletteColours.find(
                colour => colour.name === settings.solidColour
            );

            heroOverlay.style.backgroundImage = 'none';

            heroOverlay.style.backgroundColor =
                solidColour.css;

        }

        opacityValue.textContent =
            settings.overlayOpacity.toFixed(2).replace(/0$/, '');

    }


    // ==========================================
    // 13. UPDATE TYPOGRAPHY
    // ==========================================

    function updateTypography() {

        heroHeading.style.setProperty(
            '--h1-font',
            settings.h1Font
        );

        heroSubheading.style.setProperty(
            '--h2-font',
            settings.h2Font
        );

        heroHeading.style.setProperty(
            '--h1-spacing',
            settings.h1Spacing + 'px'
        );

        heroSubheading.style.setProperty(
            '--h2-spacing',
            settings.h2Spacing + 'px'
        );

        heroHeading.style.setProperty(
            '--h1-color',
            getColourCSS(settings.h1Colour)
        );

        heroSubheading.style.setProperty(
            '--h2-color',
            getColourCSS(settings.h2Colour)
        );

        h1SpacingValue.textContent =
            settings.h1Spacing + 'px';

        h2SpacingValue.textContent =
            settings.h2Spacing + 'px';

    }


    // ==========================================
    // 14. GET CSS COLOUR
    // ==========================================

    function getColourCSS(colourName) {

        const colour = paletteColours.find(
            item => item.name === colourName
        );

        return colour ? colour.css : 'var(--color-white)';

    }


    // ==========================================
    // 15. UPDATE PALETTE MODE
    // ==========================================

    function updatePaletteMode() {

        if (settings.gradientOverlay) {

            solidTargets.hidden = true;
            gradientTargets.hidden = false;
            directionSection.hidden = false;

            if (
                settings.activeTarget === 'overlay' ||
                settings.activeTarget === 'h2' && !settings.showSubheading
            ) {
                settings.activeTarget = 'start';
            }

        } else {

            solidTargets.hidden = false;
            gradientTargets.hidden = true;
            directionSection.hidden = true;

            if (
                settings.activeTarget === 'start' ||
                settings.activeTarget === 'end' ||
                settings.activeTarget === 'h2' && !settings.showSubheading
            ) {
                settings.activeTarget = 'overlay';
            }

        }

        updatePaletteTargets();
        updatePaletteSelection();

    }


    // ==========================================
    // 16. UPDATE PREVIEW
    // ==========================================

    function updatePreview() {

        readSettings();

        heroPreview.classList.toggle(
            'hero-centre',
            settings.centreText
        );

        heroPreview.classList.toggle(
            'hero-min-height',
            settings.minimumHeight
        );

        heroSubheading.hidden =
            !settings.showSubheading;

        h2Typography.hidden =
            !settings.showSubheading;

        updatePaletteMode();
        updateOverlay();
        updateTypography();

        updateExportTextarea();

    }


    // ==========================================
    // 17. ESCAPE HTML TEXT
    // ==========================================

    function escapeHTML(text) {

        return text
            .replace(/&/g, '&amp;')
            .replace(/</g, '&lt;')
            .replace(/>/g, '&gt;')
            .replace(/"/g, '&quot;')
            .replace(/'/g, '&#39;');

    }


    // ==========================================
    // 18. BUILD EXPORTED HERO CSS
    // ==========================================

    function buildExportStyles() {

        let css = heroStyles;

        css = css.replace(
            'outputs/assets/img/hero-image.jpg',
            exportedImagePath
        );

        return css;

    }


    // ==========================================
    // 19. BUILD STANDALONE HTML
    // ==========================================

    function buildExportHTML() {

        readSettings();

        const headingText =
            escapeHTML(heroHeading.textContent);

        const subheadingText =
            escapeHTML(heroSubheading.textContent);

        const heroClasses = [
            'hero',
            settings.centreText ? 'hero-centre' : '',
            settings.minimumHeight ? 'hero-min-height' : ''
        ].filter(Boolean).join(' ');

        const overlayClass = settings.gradientOverlay
            ? 'hero-overlay hero-overlay-gradient'
            : 'hero-overlay hero-overlay-solid';

        const overlayStyle = settings.gradientOverlay
            ? `opacity: ${settings.overlayOpacity}; background-image: linear-gradient(${settings.gradientDirection}, ${getColourCSS(settings.gradientStart)}, ${getColourCSS(settings.gradientEnd)}); background-color: transparent;`
            : `opacity: ${settings.overlayOpacity}; background-color: ${getColourCSS(settings.solidColour)}; background-image: none;`;

        const h1Style = [
            `--h1-font: ${settings.h1Font}`,
            `--h1-spacing: ${settings.h1Spacing}px`,
            `--h1-color: ${getColourCSS(settings.h1Colour)}`
        ].join('; ');

        const h2Style = [
            `--h2-font: ${settings.h2Font}`,
            `--h2-spacing: ${settings.h2Spacing}px`,
            `--h2-color: ${getColourCSS(settings.h2Colour)}`
        ].join('; ');

        let subheadingHTML = '';

        if (settings.showSubheading) {

            subheadingHTML =
                `<h2 style="${h2Style}">${subheadingText}</h2>`;

        }

        const completeHTML = `<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <title>Hero Image</title>

    <meta name="description" content="Short, unique description of web page. Often used as the snippet shown in search results. About 120 characters is ideal.">

    <link rel="preload" as="image" href="${exportedImagePath}">

    <script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>

    <style type="text/tailwindcss">
${buildExportStyles()}
    </style>
</head>

<body>

    <header class="${heroClasses}">

        <div class="hero-image" aria-hidden="true"></div>

        <div
            class="${overlayClass}"
            aria-hidden="true"
            style="${overlayStyle}"
        ></div>

        <div class="hero-content">

            <h1 style="${h1Style}">${headingText}</h1>

            ${subheadingHTML}

        </div>

    </header>

</body>
</html>`;

        return completeHTML;

    }


    // ==========================================
    // 20. UPDATE MANUAL COPY TEXTAREA
    // ==========================================

    function updateExportTextarea() {

        exportedHTML.value = buildExportHTML();

    }


    // ==========================================
    // 21. INSERT PLAIN TEXT AT CURSOR
    // ==========================================

    function insertPlainTextAtCursor(element, text) {

        const selection = window.getSelection();

        if (!selection || selection.rangeCount === 0) {

            element.appendChild(
                document.createTextNode(text)
            );

            return;

        }

        const range = selection.getRangeAt(0);

        if (!element.contains(range.commonAncestorContainer)) {

            element.appendChild(
                document.createTextNode(text)
            );

            return;

        }

        range.deleteContents();

        const textNode =
            document.createTextNode(text);

        range.insertNode(textNode);

        range.setStartAfter(textNode);
        range.collapse(true);

        selection.removeAllRanges();
        selection.addRange(range);

    }


    // ==========================================
    // 22. HANDLE PLAIN TEXT PASTE
    // ==========================================

    function handlePlainTextPaste(event) {

        event.preventDefault();

        const pastedText =
            event.clipboardData.getData('text/plain');

        insertPlainTextAtCursor(
            event.currentTarget,
            pastedText
        );

    }


    // ==========================================
    // 23. HANDLE HEADING KEYBOARD INPUT
    // ==========================================

    function handleHeadingKeydown(event) {

        if (event.key === 'Enter') {

            event.preventDefault();

            insertPlainTextAtCursor(
                event.currentTarget,
                '\n'
            );

        }

    }


    // ==========================================
    // 24. COPY HTML
    // ==========================================

    async function copyHTML() {

        const htmlToCopy = buildExportHTML();

        exportedHTML.value = htmlToCopy;

        try {

            if (
                !navigator.clipboard ||
                !navigator.clipboard.writeText
            ) {

                throw new Error('Clipboard unavailable');

            }

            await navigator.clipboard.writeText(htmlToCopy);

            copyStatus.textContent =
                'HTML copied—paste it into VS Code.';

            fallbackSection.hidden = true;

        }

        catch (error) {

            copyStatus.textContent =
                'Automatic copying is unavailable. Copy the HTML from the box below.';

            fallbackSection.hidden = false;

            exportedHTML.focus();
            exportedHTML.select();

        }

    }
